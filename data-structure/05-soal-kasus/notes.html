Data structure · 05

Mengerjakan soal kasus

Cara mengubah soal cerita menjadi program C lengkap yang memakai linked list, stack, queue, atau priority queue.

Slide · Materi yang dipakai: 01, 02, 03, 04

TL;DR
  • Resep: format I/O → pilih struktur → petakan kalimat ke operasi → kerangka → trace sample → edge case.
  • Struktur dipilih dari pertanyaan "siapa yang keluar duluan?": urutan datang (queue), yang terbaru (stack), atau prioritas (priority queue).
  • Fungsi operasi diambil dari materi 01 sampai 04. main hanya membaca input, memanggil fungsi sesuai urutan soal, dan mencetak.
  • Soal biasanya dicek manual: data harus benar-benar diproses lewat struktur yang diminta, bukan langsung dicetak.

1. Resep 6 langkah

  1. Baca format input dan output. Tulis sample input di kertas, lalu tandai arti setiap angka.
  2. Pilih struktur dengan bertanya: siapa yang keluar atau dilayani duluan?
  3. Petakan setiap kalimat soal ke satu operasi (satu fungsi).
  4. Tulis kerangka: struct, createNode, fungsi operasi, lalu main.
  5. Trace sample input dengan tangan, lalu cocokkan dengan sample output.
  6. Cek edge case dan aturan soal, misalnya "wajib memakai linked list" atau "list B harus kosong".

Langkah 1 sampai 3 dikerjakan sebelum menulis kode. Kalau ketiganya jelas, menulis kodenya tinggal menyusun fungsi yang sudah kamu kenal.

Soal. Dua lajur tol, A dan B, menyatu menjadi satu lajur. Lajur A berisi N mobil dan lajur B berisi M mobil, masing-masing dengan nomor unik. #include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

typedef struct {
    Node* head;
    Node* tail;
} Queue;

Node* createNode(int x) {
    Node* n = (Node*)malloc(sizeof(Node));
    n->data = x;
    n->next = NULL;
    return n;
}

void enqueue(Queue* q, int x) {
    Node* newNode = createNode(x);
    if (q->tail == NULL) {
        q->head = q->tail = newNode;
    } else {
        q->tail->next = newNode;
        q->tail = newNode;
    }
}

int dequeue(Queue* q) {              // syarat: q->head != NULL
    Node* del = q->head;
    int x = del->data;
    q->head = del->next;
    if (q->head == NULL)
        q->tail = NULL;
    free(del);
    return x;
}

void cancel(Queue* q, int x) {
    Node* prev = NULL;
    Node* cur = q->head;
    while (cur != NULL && cur->data != x) {
        prev = cur;
        cur = cur->next;
    }
    if (cur == NULL) return;
    if (prev == NULL)
        q->head = cur->next;
    else
        prev->next = cur->next;
    if (cur == q->tail)
        q->tail = prev;
    free(cur);
}

void printQueue(Queue* q) {
    for (Node* cur = q->head; cur != NULL; cur = cur->next) {
        if (cur != q->head) printf(" ");
        printf("%d", cur->data);
    }
    printf("\n");
}

int main() {
    Queue a = {NULL, NULL}, b = {NULL, NULL}, merged = {NULL, NULL};
    int n, m, l, x;

    scanf("%d %d", &n, &m);
    for (int i = 0; i < n; i++) {
        scanf("%d", &x);
        enqueue(&a, x);
    }
    for (int i = 0; i < m; i++) {
        scanf("%d", &x);
        enqueue(&b, x);
    }

    while (a.head != NULL || b.head != NULL) {     // bergantian, mulai dari A
        if (a.head != NULL) enqueue(&merged, dequeue(&a));
        if (b.head != NULL) enqueue(&merged, dequeue(&b));
    }

    scanf("%d", &l);
    for (int i = 0; i < l; i++) {
        scanf("%d", &x);
        cancel(&merged, x);
    }

    printQueue(&merged);
    while (merged.head != NULL)
        dequeue(&merged);
    return 0;
}

5. Contoh 2: sambung playlist (double linked list)

Soal. Playlist A berisi N lagu dan playlist B berisi M lagu (nomor lagu, N dan M boleh 0). Pindahkan semua lagu B ke belakang A, sehingga B menjadi kosong. Cetak A dari depan, lalu dari belakang. Wajib memakai double linked list.

InputOutput
3 2
4 1 2
7 9
4 1 2 7 9
9 7 2 1 4

Menurunkan append dengan resep materi 01

  1. Gambar. A: [4] ⇄ [1] ⇄ [2], B: [7] ⇄ [9]. Sesudah: [4] ⇄ [1] ⇄ [2] ⇄ [7] ⇄ [9], B kosong.
  2. Panah baru: satu sambungan dua arah [2] ⇄ [7], jadi 2 baris. Pointer list juga berubah: a->tail pindah ke [9], dan B menjadi kosong.
  3. Pemilik panah: [2] adalah a->tail dan [7] adalah b->head. Keduanya langsung tersedia, jadi tidak perlu traverse: O(1).
  4. Urutan: a->tail dipakai untuk menyambung dulu, baru dipindah ke b->tail.
  5. Edge case: B kosong (tidak ada yang berubah), A kosong (tidak ada a->tail, jadi A langsung memakai head B).
void append(DList* a, DList* b) {         // pindahkan semua node B ke belakang A
    if (b->head == NULL) return;          // B kosong: A tidak berubah
    if (a->head == NULL) {
        a->head = b->head;                // A kosong: A langsung jadi B
    } else {
        a->tail->next = b->head;          // sambungan dua arah
        b->head->prev = a->tail;
    }
    a->tail = b->tail;
    b->head = b->tail = NULL;             // B kosong, node-nya sekarang milik A
}
  • Kenapa B wajib dikosongkan? Node B sekarang milik A. Kalau B masih menunjuk node itu, free B nanti melepas node yang masih dipakai A (double free).
  • Cetak maju dan mundur. Kalau hasil mundur tidak terbalik, ada prev yang lupa disambung.
Lihat full code contoh 2 
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* prev;
    struct Node* next;
} Node;

typedef struct {
    Node* head;
    Node* tail;
} DList;

Node* createNode(int x) {
    Node* n = (Node*)malloc(sizeof(Node));
    n->data = x;
    n->prev = n->next = NULL;
    return n;
}

void insertBack(DList* l, int x) {
    Node* newNode = createNode(x);
    if (l->tail == NULL) {
        l->head = l->tail = newNode;
    } else {
        newNode->prev = l->tail;
        l->tail->next = newNode;
        l->tail = newNode;
    }
}

void append(DList* a, DList* b) {         // pindahkan semua node B ke belakang A
    if (b->head == NULL) return;          // B kosong: A tidak berubah
    if (a->head == NULL) {
        a->head = b->head;                // A kosong: A langsung jadi B
    } else {
        a->tail->next = b->head;          // sambungan dua arah
        b->head->prev = a->tail;
    }
    a->tail = b->tail;
    b->head = b->tail = NULL;             // B kosong, node-nya sekarang milik A
}

void printForward(DList* l) {
    for (Node* cur = l->head; cur != NULL; cur = cur->next) {
        if (cur != l->head) printf(" ");
        printf("%d", cur->data);
    }
    printf("\n");
}

void printBackward(DList* l) {
    for (Node* cur = l->tail; cur != NULL; cur = cur->prev) {
        if (cur != l->tail) printf(" ");
        printf("%d", cur->data);
    }
    printf("\n");
}

void freeList(DList* l) {
    while (l->head != NULL) {
        Node* del = l->head;
        l->head = del->next;
        free(del);
    }
    l->tail = NULL;
}

int main() {
    DList a = {NULL, NULL}, b = {NULL, NULL};
    int n, m, x;

    scanf("%d %d", &n, &m);
    for (int i = 0; i < n; i++) {
        scanf("%d", &x);
        insertBack(&a, x);
    }
    for (int i = 0; i < m; i++) {
        scanf("%d", &x);
        insertBack(&b, x);
    }

    append(&a, &b);
    printForward(&a);
    printBackward(&a);       // kalau ada prev yang salah, baris ini tidak terbalik

    freeList(&a);
    return 0;
}

Latihan 1: tanda kurung (stack)

Diberikan T string berisi karakter ( ) [ ] { }, tanpa spasi, panjang paling banyak 100. Untuk setiap string, cetak YES kalau semua kurung berpasangan dengan benar, dan NO kalau tidak.

InputOutput

Petunjuk

Kurung yang terakhir dibuka harus ditutup duluan, jadi pakai stack. Kurung buka: push. Kurung tutup: stack tidak boleh kosong, dan hasil pop harus pasangannya. Di akhir, stack harus kosong. Tiga sample NO menguji tiga kegagalan berbeda.

Full code 
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

typedef struct {
    Node* top;
} Stack;

Node* createNode(int x) {
    Node* n = (Node*)malloc(sizeof(Node));
    n->data = x;
    n->next = NULL;
    return n;
}

void push(Stack* s, int x) {
    Node* newNode = createNode(x);
    newNode->next = s->top;
    s->top = newNode;
}

int pop(Stack* s) {
    Node* del = s->top;
    int x = del->data;
    s->top = del->next;
    free(del);
    return x;
}

char openOf(char close) {            // pasangan buka dari sebuah kurung tutup
    if (close == ')') return '(';
    if (close == ']') return '[';
    return '{';
}

int balanced(char str[]) {
    Stack s = {NULL};
    int ok = 1;
    for (int i = 0; str[i] != '\0' && ok; i++) {
        char c = str[i];
        if (c == '(' || c == '[' || c == '{')
            push(&s, c);
        else if (s.top == NULL || pop(&s) != openOf(c))
            ok = 0;                  // tutup tanpa pasangan, atau pasangannya salah
    }
    if (s.top != NULL) ok = 0;       // masih ada kurung buka yang belum ditutup
    while (s.top != NULL)            // free sisa node
        pop(&s);
    return ok;
}

int main() {
    int t;
    char str[101];
    scanf("%d", &t);
    for (int i = 0; i < t; i++) {
        scanf("%s", str);
        printf("%s\n", balanced(str) ? "YES" : "NO");
    }
    return 0;
}

Latihan 2: hitung sampai K (queue)

N anak bernomor 1 sampai N duduk melingkar. Mulai dari anak 1, mereka berhitung 1, 2, ..., K. Anak yang menyebut K keluar, lalu anak sesudahnya mulai berhitung lagi dari 1. Cetak urutan anak yang keluar.

InputOutput
5 2
2 4 1 5 3
7 3
3 6 2 7 5 1 4
Petunjuk

Lingkaran bisa disimulasikan dengan queue: anak yang tidak keluar pindah dari depan ke belakang dengan enqueue(&q, dequeue(&q)). Ulangi K - 1 kali, lalu dequeue anak ke-K dan cetak.

Full code 
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

typedef struct {
    Node* head;
    Node* tail;
} Queue;

Node* createNode(int x) {
    Node* n = (Node*)malloc(sizeof(Node));
    n->data = x;
    n->next = NULL;
    return n;
}

void enqueue(Queue* q, int x) {
    Node* newNode = createNode(x);
    if (q->tail == NULL) {
        q->head = q->tail = newNode;
    } else {
        q->tail->next = newNode;
        q->tail = newNode;
    }
}

int dequeue(Queue* q) {
    Node* del = q->head;
    int x = del->data;
    q->head = del->next;
    if (q->head == NULL)
        q->tail = NULL;
    free(del);
    return x;
}

int main() {
    Queue q = {NULL, NULL};
    int n, k;
    scanf("%d %d", &n, &k);
    for (int i = 1; i <= n; i++)
        enqueue(&q, i);

    int first = 1;
    while (q.head != NULL) {
        for (int i = 0; i < k - 1; i++)          // k-1 anak pindah ke belakang
            enqueue(&q, dequeue(&q));
        if (!first) printf(" ");
        printf("%d", dequeue(&q));              // anak ke-k keluar
        first = 0;
    }
    printf("\n");
    return 0;
}

Latihan 3: antrean vaksin (priority queue dengan aturan berlapis)

N orang datang berurutan, masing-masing dengan nama dan umur. Lansia (umur 60 ke atas) dilayani lebih dulu. Sesama lansia, yang lebih tua dilayani lebih dulu, dan kalau umurnya sama, yang datang duluan. Yang bukan lansia dilayani sesuai urutan datang. Cetak urutan layanan.

Tulis higher(a, b) dari materi 04. Aturan 1: lansia sebelum bukan lansia. Aturan 2: sesama lansia, umur lebih besar duluan. Selain itu return 0, supaya urutan datang dipertahankan.

Full code 
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct Node {
    char name[21];
    int age;
    struct Node* next;
} Node;

typedef struct {
    Node* head;
} PQueue;

Node* createNode(char name[], int age) {
    Node* n = (Node*)malloc(sizeof(Node));
    strcpy(n->name, name);
    n->age = age;
    n->next = NULL;
    return n;
}

// return 1 kalau a harus dilayani sebelum b
int higher(Node* a, Node* b) {
    int oldA = a->age >= 60, oldB = b->age >= 60;
    if (oldA != oldB) return oldA;          // lansia duluan
    if (oldA && a->age != b->age)
        return a->age > b->age;             // sesama lansia: yang lebih tua duluan
    return 0;                               // selain itu: urutan datang
}

void pqPush(PQueue* pq, Node* newNode) {
    if (pq->head == NULL || higher(newNode, pq->head)) {
        newNode->next = pq->head;
        pq->head = newNode;
        return;
    }
    Node* cur = pq->head;
    while (cur->next != NULL && !higher(newNode, cur->next))
        cur = cur->next;
    newNode->next = cur->next;
    cur->next = newNode;
}

int main() {
    PQueue pq = {NULL};
    int n, age;
    char name[21];
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        scanf("%s %d", name, &age);
        pqPush(&pq, createNode(name, age));
    }
    while (pq.head != NULL) {
        Node* del = pq.head;
        printf("%s\n", del->name);
        pq.head = del->next;
        free(del);
    }
    return 0;
}

Latihan 4: balik K angka pertama (queue dan stack)

Diberikan antrean berisi N angka. Balik urutan K angka pertama, sisanya tetap. Wajib memakai queue dan stack dengan linked list.

InputOutput
5 3
1 2 3 4 5
3 2 1 4 5
Petunjuk

Tiga langkah: (1) dequeue K angka dan push ke stack, (2) pop semua dan enqueue kembali, sehingga K angka terbalik ada di belakang, (3) pindahkan N - K angka yang ada di depan ke belakang. Trace dengan sample: 1 2 3 4 5, lalu 4 5 3 2 1, lalu 3 2 1 4 5.

Full code 
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

typedef struct {
    Node* top;
} Stack;

typedef struct {
    Node* head;
    Node* tail;
} Queue;

Node* createNode(int x) {
    Node* n = (Node*)malloc(sizeof(Node));
    n->data = x;
    n->next = NULL;
    return n;
}

/* ---------- Stack: masuk dan keluar di top ---------- */
void push(Stack* s, int x) {
    Node* newNode = createNode(x);
    newNode->next = s->top;
    s->top = newNode;
}

int pop(Stack* s) {                  // syarat: s->top != NULL
    Node* del = s->top;
    int x = del->data;
    s->top = del->next;
    free(del);
    return x;
}

/* ---------- Queue: masuk di tail, keluar di head ---------- */
void enqueue(Queue* q, int x) {
    Node* newNode = createNode(x);
    if (q->tail == NULL) {
        q->head = q->tail = newNode;
    } else {
        q->tail->next = newNode;
        q->tail = newNode;
    }
}

int dequeue(Queue* q) {              // syarat: q->head != NULL
    Node* del = q->head;
    int x = del->data;
    q->head = del->next;
    if (q->head == NULL)
        q->tail = NULL;
    free(del);
    return x;
}

int main() {
    Queue q = {NULL, NULL};
    Stack s = {NULL};
    int n, k, x;
    scanf("%d %d", &n, &k);
    for (int i = 0; i < n; i++) {
        scanf("%d", &x);
        enqueue(&q, x);
    }

    for (int i = 0; i < k; i++)          // 1. k elemen pertama masuk stack
        push(&s, dequeue(&q));
    while (s.top != NULL)                // 2. keluar dari stack dalam urutan terbalik,
        enqueue(&q, pop(&s));            //    masuk ke belakang queue
    for (int i = 0; i < n - k; i++)      // 3. n-k elemen sisanya pindah ke belakang
        enqueue(&q, dequeue(&q));

    while (q.head != NULL) {
        printf("%d", dequeue(&q));
        printf(q.head != NULL ? " " : "\n");
    }
    return 0;
}

Latihan 5: menggabungkan dua double linked list (soal ASG)

Diberikan dua double linked list, A berisi N bilangan dan B berisi M bilangan. Gabungkan B ke belakang A, sehingga A berisi N + M bilangan dan B menjadi kosong. Cetak isi A setelah digabung dalam satu baris.

Input: baris 1 berisi N dan M (positif), baris 2 berisi N bilangan untuk A, baris 3 berisi M bilangan untuk B. Wajib: masukkan bilangan ke A dan B sesuai urutan, lalu benar-benar gabungkan list-nya. Mencetak bilangan tanpa menggabungkan list dianggap salah, karena kode dicek manual.

InputOutput
3 5
4 1 2
2 1 5 2 3
4 1 2 2 1 5 2 3
Petunjuk

Ini contoh 2 (sambung playlist) tanpa cetak mundur. Pakai append: satu sambungan dua arah antara tail A dan head B, lalu B dikosongkan.

Full code 
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* prev;
    struct Node* next;
} Node;

typedef struct {
    Node* head;
    Node* tail;
} DList;

Node* createNode(int x) {
    Node* n = (Node*)malloc(sizeof(Node));
    n->data = x;
    n->prev = n->next = NULL;
    return n;
}

void insertBack(DList* l, int x) {
    Node* newNode = createNode(x);
    if (l->tail == NULL) {
        l->head = l->tail = newNode;
    } else {
        newNode->prev = l->tail;
        l->tail->next = newNode;
        l->tail = newNode;
    }
}

void append(DList* a, DList* b) {         // pindahkan semua node B ke belakang A
    if (b->head == NULL) return;          // B kosong: A tidak berubah
    if (a->head == NULL) {
        a->head = b->head;                // A kosong: A langsung jadi B
    } else {
        a->tail->next = b->head;          // sambungan dua arah
        b->head->prev = a->tail;
    }
    a->tail = b->tail;
    b->head = b->tail = NULL;             // B kosong, node-nya sekarang milik A
}

void printForward(DList* l) {
    for (Node* cur = l->head; cur != NULL; cur = cur->next) {
        if (cur != l->head) printf(" ");
        printf("%d", cur->data);
    }
    printf("\n");
}


void freeList(DList* l) {
    while (l->head != NULL) {
        Node* del = l->head;
        l->head = del->next;
        free(del);
    }
    l->tail = NULL;
}

int main() {
    DList a = {NULL, NULL}, b = {NULL, NULL};
    int n, m, x;

    scanf("%d %d", &n, &m);
    for (int i = 0; i < n; i++) {
        scanf("%d", &x);
        insertBack(&a, x);
    }
    for (int i = 0; i < m; i++) {
        scanf("%d", &x);
        insertBack(&b, x);
    }

    append(&a, &b);
    printForward(&a);       // b sekarang kosong: b.head == NULL

    freeList(&a);
    return 0;
}

Latihan 6: fair waiting list (soal quiz)

Full code 
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node* next;
} Node;

typedef struct {
    Node* head;
    Node* tail;
} Queue;

Node* createNode(int x) {
    Node* n = (Node*)malloc(sizeof(Node));
    n->data = x;
    n->next = NULL;
    return n;
}

void enqueue(Queue* q, int x) {
    Node* newNode = createNode(x);
    if (q->tail == NULL) {
        q->head = q->tail = newNode;
    } else {
        q->tail->next = newNode;
        q->tail = newNode;
    }
}

int dequeue(Queue* q) {              // syarat: q->head != NULL
    Node* del = q->head;
    int x = del->data;
    q->head = del->next;
    if (q->head == NULL)
        q->tail = NULL;
    free(del);
    return x;
}

void cancel(Queue* q, int x) {
    Node* prev = NULL;
    Node* cur = q->head;
    while (cur != NULL && cur->data != x) {
        prev = cur;
        cur = cur->next;
    }
    if (cur == NULL) return;
    if (prev == NULL)
        q->head = cur->next;
    else
        prev->next = cur->next;
    if (cur == q->tail)
        q->tail = prev;
    free(cur);
}

void printQueue(Queue* q) {
    for (Node* cur = q->head; cur != NULL; cur = cur->next) {
        if (cur != q->head) printf(" ");
        printf("%d", cur->data);
    }
    printf("\n");
}

int main() {
    Queue online = {NULL, NULL}, walkIn = {NULL, NULL}, merged = {NULL, NULL};
    int n, m, l, x;

    scanf("%d %d", &n, &m);
    for (int i = 0; i < n; i++) {
        scanf("%d", &x);
        enqueue(&online, x);
    }
    for (int i = 0; i < m; i++) {
        scanf("%d", &x);
        enqueue(&walkIn, x);
    }

    while (online.head != NULL || walkIn.head != NULL) {     // bergantian, mulai dari online
        if (online.head != NULL) enqueue(&merged, dequeue(&online));
        if (walkIn.head != NULL) enqueue(&merged, dequeue(&walkIn));
    }

    scanf("%d", &l);
    for (int i = 0; i < l; i++) {
        scanf("%d", &x);
        cancel(&merged, x);
    }

    printQueue(&merged);
    while (merged.head != NULL)
        dequeue(&merged);
    return 0;
}
